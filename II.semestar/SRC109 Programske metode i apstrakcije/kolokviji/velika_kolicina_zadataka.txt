1. Napisati funkciju koja za dva unesena znaka vraća koliko daleko se nalaze u engleskom alfabetu (udaljenost je pozitivan broj). Ako jedan od znakova nije slovo, funkcija vraca -1. Napomena: Za provjeru je li znak slovo možete se koristiti funkcijom int isalpha(int c); iz ctype.h, a za pretvaranje znaka u veliko slovo funkciju int toupper(int c);
#include <stdio.h>
#include <ctype.h>
#include <stdlib.h>

int udaljenost(char a, char b){
    if(!isalpha(a) || !isalpha(b)) return -1;
    return abs(toupper(a)-toupper(b));
}
// isalpha provjerava slovo, toupper pretvara u veliko slovo, abs vraća pozitivnu razliku

int main(){
    printf("%d\n",udaljenost('a','d')); // 3
    printf("%d\n",udaljenost('A','z')); // 25
    printf("%d\n",udaljenost('a','1')); // -1
    return 0;
}

#include <stdio.h>

int kalkulator(int a,int b,char op){
    switch(op){
        case '+': return a+b;
        case '-': return a-b;
        case '*': return a*b;
        case '/': return b!=0?a/b:0;
        default: return 0;
    }
}
// switch izvršava operaciju ovisno o znaku

int main(){
    printf("%d\n",kalkulator(10,5,'+')); // 15
    printf("%d\n",kalkulator(10,5,'-')); // 5
    printf("%d\n",kalkulator(10,5,'*')); // 50
    printf("%d\n",kalkulator(10,5,'/')); // 2
    return 0;
}

2. Napisati funkciju koja za unesene prirodne brojeve računa razliku najvećeg i najmanjeg broja. Brojeve unosi korisnik sve dok ne unese 0.
#include <stdio.h>

// Funkcija unosi n brojeva i pronalazi maksimum i minimum
// koristeći rekurziju. Nakon unosa svih brojeva, vraća
// razliku maksimum-minimum.
void max_min_razlika(int n, int *max, int *min){
    if(n==0) return;                     // baza: nema više brojeva
    int x;
    scanf("%d",&x);
    if(n==1){                            // prvi uneseni broj inicijalizira max i min
        *max=x;
        *min=x;
    }
    max_min_razlika(n-1, max, min);      // rekurzivni poziv za preostale brojeve
    if(x>*max) *max=x;                   // ažuriranje maksimuma
    if(x<*min) *min=x;                   // ažuriranje minimuma
}

int main(){
    int n=5;                             
    int max,min;
    printf("Unesite %d prirodnih brojeva:\n", n);
    max_min_razlika(n,&max,&min);
    printf("Razlika najveceg i najmanjeg: %d\n", max-min);
    return 0;
}
// Main traži od korisnika unos 5 brojeva i ispisuje razliku

int prost(int x){
    if(x<2) return 0;
    for(int i=2;i*i<=x;i++)
        if(x%i==0) return 0;
    return 1;
}

5. Napisati funkciju koja za uneseni broj n provjerava je li on trimorfan, tj. završava li njegov kub istim znamenkama.
#include <stdio.h>

// Funkcija provjerava trimorfnost broja
// Računa kub broja i uspoređuje posljednje znamenke sa samim brojem
int trimorfan(int n){
    long long k=(long long)n*n*n; // kub broja
    int m=n;
    while(m>0){
        if(m%10 != k%10) return 0;   // ako se znamenke ne poklapaju, nije trimorfan
        m/=10;
        k/=10;
    }
    return 1;
}

int main(){
    int n;
    printf("Unesite brojeve (0 za kraj):\n");
    while(1){
        scanf("%d",&n);
        if(n==0) break;
        if(trimorfan(n)) printf("%d je trimorfan\n", n);
        else printf("%d nije trimorfan\n", n);
    }
    return 0;
}
// Main omogućuje unos više brojeva i provjeru trimorfnosti

#include <stdlib.h>

double* ubaci_aritmeticke_sredine(int* niz, int vel, int* nova_vel) {
    *nova_vel = vel*2 - 1; // nova veličina niza nakon umetanja sredina
    double* novi = (double*)malloc((*nova_vel)*sizeof(double)); // alociranje memorije
    for(int i = 0; i < vel-1; i++) {

    double b;

double povrsina(Trokut t) {

    for(int i=1;i<vel;i++){
        double opseg = niz[i].a + niz[i].b + niz[i].c;

            *maxP = niz[i];

        if(opseg > maxOpseg){

}

} Razlomak;

// Funkcija za izračunavanje najvećeg zajedničkog djelitelja
int gcd(int a, int b){
    while(b!=0){
        int t=b;
        b=a%b;
        a=t;
    }
    return a;
}

// Funkcija koja skraćuje razlomak
Razlomak skrati(Razlomak r){
    int g=gcd(r.brojnik, r.nazivnik);
    r.brojnik/=g;
    r.nazivnik/=g;
    return r;
}

// Funkcije za osnovne operacije
Razlomak zbroj(Razlomak r1, Razlomak r2){
    Razlomak r;
    r.brojnik = r1.brojnik*r2.nazivnik + r2.brojnik*r1.nazivnik;
    r.nazivnik = r1.nazivnik*r2.nazivnik;
    return skrati(r);
}

Razlomak razlika(Razlomak r1, Razlomak r2){
    Razlomak r;
    r.brojnik = r1.brojnik*r2.nazivnik - r2.brojnik*r1.nazivnik;
    r.nazivnik = r1.nazivnik*r2.nazivnik;
    return skrati(r);
}

Razlomak produkt(Razlomak r1, Razlomak r2){
    Razlomak r;
    r.brojnik = r1.brojnik*r2.brojnik;
    r.nazivnik = r1.nazivnik*r2.nazivnik;
    return skrati(r);
}

Razlomak kolicnik(Razlomak r1, Razlomak r2){
    Razlomak r;
    r.brojnik = r1.brojnik*r2.nazivnik;
    r.nazivnik = r1.nazivnik*r2.brojnik;
    return skrati(r);
}

int main(){
    Razlomak r1 = {1,2}, r2={3,4};
    Razlomak z=r1, o=r2;
    Razlomak sum=zbroj(r1,r2);
    Razlomak diff=razlika(r1,r2);
    Razlomak prod=produkt(r1,r2);
    Razlomak quot=kolicnik(r1,r2);
    printf("Zbroj: %d/%d\n", sum.brojnik, sum.nazivnik);
    printf("Razlika: %d/%d\n", diff.brojnik, diff.nazivnik);
    printf("Produkt: %d/%d\n", prod.brojnik, prod.nazivnik);
    printf("Količnik: %d/%d\n", quot.brojnik, quot.nazivnik);
    return 0;
}

19. Definirati strukturu stol. Stol je odred̄en sa svojom duljinom, širinom i visinom. Napisati funkciju koja za niz stolova vraća one stolove koji imaju najveću pisaću površinu. Napomena: Funkcija treba, osim niza stolova, vratiti i duljinu novonastalog niza stolova.

#include <stdio.h>

typedef struct {
    double duljina;
    double sirina;
    double visina;
} Stol;

// Funkcija koja vraća stolove s najvećom pisaćom površinom
int stolovi_najveca_povrs(Stol* niz, int vel, Stol* rezultati){
    double maxPovrs = niz[0].duljina*niz[0].sirina;
    rezultati[0] = niz[0];
    int count = 1;
    for(int i=1;i<vel;i++){
        double povrs = niz[i].duljina*niz[i].sirina;
        if(povrs > maxPovrs){
            maxPovrs = povrs;
            rezultati[0] = niz[i];
            count = 1;
        } else if(povrs == maxPovrs){
            rezultati[count++] = niz[i];
        }
    }
    return count;
}

int main(){
    Stol niz[]={{120,60,75},{100,50,70},{120,60,80}};
    int vel = sizeof(niz)/sizeof(niz[0]);
    Stol rezultati[vel];
    int cnt = stolovi_najveca_povrs(niz, vel, rezultati);
    printf("Stolovi s najvećom pisaćom površinom:\n");
    for(int i=0;i<cnt;i++)
        printf("D=%.2f, Š=%.2f, V=%.2f\n", rezultati[i].duljina, rezultati[i].sirina, rezultati[i].visina);
    return 0;
}
